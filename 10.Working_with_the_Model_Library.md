# Working with the Model Library

## Understanding What's Loaded

```bash
# Run the model to load it
echo "Hi there!" | ollama run granite3.3:2b

# Run the ps command
ollama ps
```

```bash
NAME             ID              SIZE      PROCESSOR    CONTEXT    UNTIL              
granite3.3:2b    07bd1f170855    1.9 GB    100% CPU     4096       4 minutes from now
```

## Inspecting Models

### Using the CLI

```bash
# this is the model we used
export MODEL="granite3.3:2b"

# print the model's metadata
ollama show $MODEL
```

```text
  Model
    architecture        granite    
    parameters          2.5B       
    context length      131072     
    embedding length    2048       
    quantization        Q4_K_M     

  Capabilities
    completion    
    tools         

  License
    Apache License               
    Version 2.0, January 2004  
```

```bash
# the Modelfile used to build this model
ollama show $MODEL --modelfile    

# default sampling parameters
ollama show $MODEL --parameters   

# the prompt template
ollama show $MODEL --template     

# the system prompt, if set
ollama show $MODEL --system
```

### Using the API

```bash
# Choose a model
export MODEL=llama3.2:3b

# Pull the model if you haven't already
ollama pull $MODEL

# Show details
curl -s http://localhost:11434/api/show \
  -d "{\"model\": \"$MODEL\"}" | jq .
```

```json
{
  "license": "Apache License...",
  "modelfile": "# Modelfile generated by ...",
  "parameters": "stop \"<|end_of_text|>\"\nstop \"<|eom_id|>\"",
  "template": "{{ .System }}{{ .Prompt }}",
  "system": "",
  "details": {
    "parent_model": "",
    "format": "gguf",
    "family": "granite",
    "families": ["granite"],
    "parameter_size": "2.5B",
    "quantization_level": "Q4_K_M"
  },
  "model_info": {
    "general.architecture": "granite",
    "general.parameter_count": 2533539840,
    "granite.context_length": 131072,
    "granite.embedding_length": 2048,
    "granite.attention.head_count": 32,
    "granite.attention.head_count_kv": 8,
    "tokenizer.ggml.model": "gpt2"

```

```bash
curl -s http://localhost:11434/api/show \
  -d '{"model": "'"$MODEL"'"}' | \
  jq '.model_info | with_entries(select(.key | endswith("context_length"))) | to_entries[0].value'
```

```bash
curl -s http://localhost:11434/api/show \
  -d '{"model": "'"$MODEL"'"}' | \
  jq -r '.capabilities[]'
```

```bash
curl -s http://localhost:11434/api/show \
  -d '{"model": "'"$MODEL"'"}' | \
  jq -r '"\(.details.family) \(.details.parameter_size) \(.details.quantization_level) caps=\(.capabilities|join(","))"'
```

```bash
# Pull three models via the API
for MODEL in llama3.2:3b granite3.3:2b llava:7b; do
  echo "Pulling $MODEL..."
  curl -s http://localhost:11434/api/pull \
    -d "{\"model\": \"$MODEL\"}" > /dev/null
done
```

```bash
# Grab the list of installed model names
MODELS=$(curl -s http://localhost:11434/api/tags | jq -r '.models[].name')

# Header, then one row per model
printf "MODEL\tSIZE\tQUANT\tCAPABILITIES\n"

# Loop through models and print a summary line for each
for MODEL in $MODELS; do
  INFO=$(curl -s http://localhost:11434/api/show -d "{\"model\": \"$MODEL\"}")
  SIZE=$(echo "$INFO"  | jq -r '.details.parameter_size')
  QUANT=$(echo "$INFO" | jq -r '.details.quantization_level')
  CAPS=$(echo "$INFO"  | jq -r '.capabilities | join(",")')
  printf "%s\t%s\t%s\t%s\n" "$MODEL" "$SIZE" "$QUANT" "$CAPS"
done | column -t -s $'\t'
```

## Stopping a Loaded Model

```bash
ollama stop $MODEL
```

## Removing a Model

```bash
MODEL=llama3.2:3b
ollama rm $MODEL
```

```bash
# The model schema will be gone from the list of models on disk
ls -lrth \
  $HOME/.ollama/models/manifests/registry.ollama.ai/library/

# Or
# ls -lrth \
#  /usr/share/ollama/.ollama/models/manifests/registry.ollama.ai/library/

# Its weights will also be gone
ls -lrth $HOME/.ollama/models/blobs/

# Or
# ls -lrth \
#  /usr/share/ollama/.ollama/models/blobs/
```

```bash
MODELS="granite3.3:2b llama3.2:3b"

ollama rm $MODELS
```

## Copying a Model

```bash
ollama cp $SOURCE $TARGET
```

```bash
# re-download granite if you deleted it previously
MODEL=granite3.3:2b
ollama pull $MODEL

# copy it
ollama cp $MODEL my-granite-tuned
```
